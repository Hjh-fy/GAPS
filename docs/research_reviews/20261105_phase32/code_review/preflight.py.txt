class Tiny64(torch.nn.Module):
    def __init__(self):
        super().__init__();self.encoder=torch.nn.Linear(8,64);self.classifier=torch.nn.Linear(64,4)
    def forward(self,x):
        z=torch.tanh(self.encoder(x[:,0,:]));return self.classifier(z),torch.nn.functional.normalize(z,dim=1),z
def ds(n,tag):
