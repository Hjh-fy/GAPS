    path=Path(path)
    if set(lock)!={'kind','sha256','source_identity_sha256','seed','shape','labels','initial_sha256','round_receipts'}:raise ValueError('unknown checkpoint contract')
    if lock['kind']!=kind or lock['seed']!=42 or lock['shape']!=[50,8] or lock['labels']!=spec('SOURCE')['labels']:raise ValueError('model identity')
    if lock['source_identity_sha256']!=source_identity_sha() or sha(path)!=lock['sha256']:raise ValueError('checkpoint provenance SHA')
    registered=HERE/'contracts/SOURCE_RANDOM_INITIAL_LOCK.json'
    if sha(registered)!='49bd47670e77b1c934940198183fdc1f0779b75e6ad9c00d3dc74d9cb896ba83':raise ValueError('registered random lock altered')
    anchor=read(registered)
    if sha(HERE/'contracts/SOURCE_RANDOM_INITIAL.pth')!=anchor['sha256'] or lock['initial_sha256']!=anchor['initial_sha256']:raise ValueError('initial ancestor mismatch')
    if kind=='sstar':
        rows=lock['round_receipts']
        if len(rows)!=25 or [r['round'] for r in rows]!=list(range(1,26)):raise ValueError('source round ancestry')
        previous=lock['initial_sha256']
        for r in rows:
            if r['received']!=previous or sorted(r['clients'])!=[1,2] or r['samples']!=[840,1680]:raise ValueError('source ancestry chain')
            receipt=path.parent/f"phase32_receipt_{r['round']:03}.json";checkpoint=path.parent/f"server_round_{r['round']:03}.pth"
            if not receipt.is_file() or read(receipt)!=r or not checkpoint.is_file() or sha(checkpoint)!=r.get('checkpoint_sha256'):raise ValueError('missing source round artifacts')

class FixedBatches:
    def __init__(self,block):
        verify_prior()
        if block not in range(1,26):raise ValueError('round outside fixed endpoint')
        schedule=prior.rows(P31/'binding/PROPOSED_TARGET_BATCH_SCHEDULE.csv')
        self.rows=[[int(i) for i in r['indices'].split(';')] for r in schedule if int(r['block'])==block]
        if len(self.rows)!=100 or any(len(r)!=16 for r in self.rows):raise ValueError('sampler contract')
    def __iter__(self):return iter(self.rows)
    def __len__(self):return 100
class RecordedLoader:
    def __init__(self,loader,audit,block,indices=None):self.loader=loader;self.dataset=loader.dataset;self.audit=audit;self.block=block;self.indices=indices;self.batch_size=loader.batch_size;self.drop_last=loader.drop_last
    def __len__(self):return len(self.loader)
    def __iter__(self):
        for i,batch in enumerate(self.loader):
            r=dict(block=self.block,batch=i,rows=len(batch[0]))
            if self.indices is not None:r['indices']=list(self.indices.rows[i])
            self.audit.append(r);yield batch
def target_loader(dataset,block,audit):
    import torch
    sampler=FixedBatches(block);loader=torch.utils.data.DataLoader(dataset,batch_sampler=sampler,num_workers=0)
    return RecordedLoader(loader,audit,block,sampler)
