    p1.verify_sha(Path(__file__).with_name('study.py'),read(RUN/'GATE1_RECOVERY.json')['code_sha256'])

            assert len(gg)==21;g=int(gg[0]['gas_id']);e=np.array([float(x['prediction_'+r['mode']+'_ppm'])-float(x['truth_ppm']) for x in gg]);nse.append(float(np.mean(e*e)/SPAN[g]**2));mae.append(float(np.mean(abs(e))))
        diff=abs(100*np.sqrt(np.mean(nse))-float(r['single_window_equal_exposure_NRMSE_percent']));assert len(groups)==40 and diff<1e-10 and abs(np.mean(mae)-float(r['single_window_equal_exposure_MAE_ppm']))<1e-10;checks.append(diff)

        p1.verify_sha(RUN/r['path'],r['sha256']);res=joblib.load(RUN/r['path']);fb,fr=baseline(v,r['target'],r['budget'],r['gas_id']);assert r['params']==fr['params']
        for key in ['mean','scale']:np.testing.assert_array_equal(res['inner'][key],fb['numerical'][key])

            for _ in range(2000):
                ids=np.concatenate([rng.choice(np.flatnonzero(gas==g),10,replace=True) for g in range(4)]);boot.append(100*(np.sqrt(bb[ids].mean())-np.sqrt(aa[ids].mean())))

    keys=list(matrix[0]);text+='主指标为Oracle_ALL、单窗口误差暴露等权NRMSE%，裁剪结果如下；raw、MAE、暴露/气体/浓度/seen/unseen完整保存。\n\n| '+' | '.join(keys)+' |\n| '+' | '.join(['---']*len(keys))+' |\n'
