def fit_residual(v,alpha,prior,x,y,source,g,rho,params):
    b=alpha.base(source,g,rho);inner=v.svr.fit_head(x,y-(b-LOW[g]),g,params)
    return dict(inner=inner,rho=float(rho),gas=g,clip_min=float(min(y)),clip_max=float(max(y)))
def safe_predict(v,prior,residual,fallback,x,rho,provider,clip=False):
    # Fallback is always the independent original target head, never the residual learner.
    if rho==0:return p1.predict(v,fallback,x,clip), 'rho0_original_F'
    try:
        value=provider()
        source=np.asarray(value,dtype=float)
        if value is None or source.shape!=(len(x),) or not np.isfinite(source).all():raise ValueError('invalid source output')
    except (ValueError,TypeError,KeyError,OSError,RuntimeError,AttributeError,StopIteration):return p1.predict(v,fallback,x,clip),'source_failure_original_F'
    h=residual['inner'];g=residual['gas'];head=prior.PriorHead(h['mean'],h['scale'],h['learner'],g,float(rho),float((1-rho)*LOW[g]))
    out=prior.predict_head(head,x,source,clip=False)
    if not np.isfinite(out).all():raise ValueError('residual numerical failure')
    return (np.clip(out,residual['clip_min'],residual['clip_max']) if clip else out),'residual'
