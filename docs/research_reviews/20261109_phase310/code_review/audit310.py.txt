"""Saved-log checks shared by the preflight, inference and scoring gates."""
from collections import Counter
from pathlib import Path
import re

from common310 import lines, require, sha


def inspect_log(file, pc, expected_denials=(), complete=True):
    events = lines(file)
    require(events and events[0]["event"] == "start"
            and [e["seq"] for e in events] == list(range(1, len(events) + 1))
            and len({e["pid"] for e in events}) == 1, "Open log sequence/PID")
    if complete:
        require(events[-1]["event"] == "end" and events[-1]["status"] == "COMPLETE", "Open log closure")
    attempts = [e for e in events if e["event"] == "open_attempt"]
    outcomes = {e["call"]: e for e in events if e["event"] == "open_result"}
    api = [e["seq"] for e in events if e["event"] == "api_open"]
    require(set(api) == set(outcomes), "Wrapped open outcomes")
    deliberate = []
    errors_allowed = set()
    for item in attempts:
        call = item["call"]
        path = item["path"]
        error = outcomes.get(call, {}).get("error")
        if item["decision"] == "DENY":
            if item["stage"] == "scope_probe":
                deliberate.append(path)
            else:
                require(path in pc["optional_denials"] and item["stage"] == "initial_load",
                        f"Unexpected open denial {path}")
            require(call is None or error == "PermissionError", "Denial outcome")
            if call is not None:
                errors_allowed.add(call)
        if error == "FileNotFoundError":
            package = item["rule"] == "python_installation" and path.endswith(
                ("/python312.zip", ".dist-info/entry_points.txt", ".egg-info/entry_points.txt", ".egg-info/metadata"))
            cache = item["rule"] == "own_evidence" and re.fullmatch(r"fontlist-v\d+\.json", Path(path).name)
            if item["stage"] == "initial_load" and (package or cache):
                errors_allowed.add(call)
    require(Counter(deliberate) == Counter(expected_denials), "Scope probe count")
    require({call for call, e in outcomes.items() if e["result"] == "ERROR"} <= errors_allowed,
            "Unexplained open error")
    requests = {e["seq"]: e for e in events if e["event"] == "numeric_request"}
    allowed = [e for e in attempts if e["decision"] == "ALLOW"]
    summary = dict(pid=events[0]["pid"], open_events=len(attempts), allowed=len(allowed),
                   denied=len(attempts)-len(allowed), numeric_requests=len(requests),
                   api_errors=dict(Counter(e["error"] for e in outcomes.values() if e["result"] == "ERROR")),
                   sha256=sha(file) if complete else None)
    return summary, events, requests


def ledger_check(ledger_path, requests, expected):
    rows = lines(ledger_path)
    counts = Counter(row["kind"] for row in rows)
    require(dict(counts) == expected, "Numerical ledger budget")
    require(set(requests) == {r["request_seq"] for r in rows}, "Numerical request coverage")
    for row in rows:
        event = requests[row["request_seq"]]
        require(all(row[k] == event[k] for k in ("pid", "stage", "request")), "Numerical ledger open binding")
        if "signal_sha256" in row:
            require(row["signal_sha256"] == event["signal_sha256"], "Numerical signal digest")
    return rows
