def episode_masks(meta,episode,budget,gas):
    codes=[10,100] if budget==2 else [10,40,70,100]
    tr=[];va=[]
    for m in meta:
        c=int(m['client_id']);d=int(m['day']);g=int(m['gas_id']);k=int(m['concentration_code']);s=float(m['start_s'])
        train=(c,d)==((1,4) if episode=='E1' else (2,1))
        valid=((c,d)==(1,4) and k in (20,30,50,60,80,90)) if episode=='E1' else (c,d)==(2,7)
        tr.append(g==gas and train and k in codes and s in (60.,150.));va.append(g==gas and valid)
    return np.array(tr),np.array(va)
def fit_krr(x,y,g,params,kernel='rbf'):
    from sklearn.kernel_ridge import KernelRidge
    mean=x.mean(0);scale=x.std(0);scale[scale<1e-9]=1.;z=(x-mean)/scale;gamma=params['gamma_factor']/x.shape[1]
    k=np.exp(-gamma*((z[:,None]-z[None,:])**2).sum(2)) if kernel=='rbf' else z@z.T
    col=k.mean(0);grand=float(k.mean());kc=k-col[None,:]-k.mean(1)[:,None]+grand
    yn=(y-LOW[g])/SPAN[g];ym=float(yn.mean());m=KernelRidge(alpha=params['alpha'],kernel='precomputed').fit(kc,yn-ym)
    return dict(kind='TargetKRR71',mean=mean,scale=scale,train_z=z,gamma=gamma,k_colmean=col,k_grand=grand,coef=m.dual_coef_,ymean=ym,
        low=float(LOW[g]),span=float(SPAN[g]),gas=g,clip_min=float(min(y)),clip_max=float(max(y)),kernel=kernel)
def krr_predict(h,x):
    z=(x-h['mean'])/h['scale'];t=h['train_z']
    k=np.exp(-h['gamma']*((z[:,None]-t[None,:])**2).sum(2)) if h.get('kernel','rbf')=='rbf' else z@t.T
    kc=k-h['k_colmean'][None,:]-k.mean(1)[:,None]+h['k_grand']
    return (kc@h['coef']+h['ymean'])*h['span']+h['low']
def fit(v,family,x,y,g,params):
    return fit_krr(x,y,g,params) if family=='KRR' else p1.fit(v,'Target'+family+'71',x,y,g,params)
def predict(v,h,x,clip=False):
    p=krr_predict(h,x) if h['kind']=='TargetKRR71' else p1.predict(v,h,x,False)
    return np.clip(p,h['clip_min'],h['clip_max']) if clip else p

def rbf_gradient(z,train,coef,gamma,span):
    d=z[None,:]-train;k=np.exp(-gamma*np.sum(d*d,axis=1));return -2*gamma*span*np.sum((coef*k)[:,None]*d,axis=0)
