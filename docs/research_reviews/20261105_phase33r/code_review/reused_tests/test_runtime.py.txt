import unittest,copy,json,sys,hashlib,tempfile
import runtime as r
class RuntimeGuards(unittest.TestCase):
    def test_raw_state_dtype_rejected_before_cast(self):
        self.assertTrue(hasattr(r,'validate_state'))
        m=r.model();s=copy.deepcopy(m.state_dict());k=next(iter(s));s[k]=s[k].double()
        with self.assertRaises(ValueError):r.validate_state(m,s)
    def test_per_class_counts_not_just_total(self):
        self.assertTrue(hasattr(r,'validate_stats'))
        proto={f'{g},0':[1.]*64 for g in range(4)};counts={f'{g},0':210 for g in range(4)}
        met=dict(prototype_json=json.dumps(proto),prototype_var_json=json.dumps(proto),class_phase_counts_json=json.dumps(counts),global_feature_json=json.dumps([1.]*64),prototype_content_sha=hashlib.sha256(json.dumps(proto).encode()).hexdigest())
        r.validate_stats(met,840)
        met['class_phase_counts_json']=json.dumps({'0,0':837,'1,0':1,'2,0':1,'3,0':1})
        with self.assertRaises(ValueError):r.validate_stats(met,840)
    def test_stats_content_receipt_mismatch_rejected(self):
        proto={f'{g},0':[1.]*64 for g in range(4)}
        met=dict(prototype_json=json.dumps(proto),prototype_var_json=json.dumps(proto),class_phase_counts_json=json.dumps({f'{g},0':210 for g in range(4)}),global_feature_json=json.dumps([1.]*64),prototype_content_sha='wrong')
        with self.assertRaises(ValueError):r.validate_stats(met,840)
    def test_actual_pure_CE_client_upload_validates(self):
        from preflight import Tiny64,ds
        r.torch.set_num_threads(1);m=Tiny64();root=r.c.HERE/'fixtures';root.mkdir(exist_ok=True);out=tempfile.mkdtemp(dir=root)
        cl=r.PhaseClient(1,r.c.spec('B4'),m,ds(840,1),out);aa=[x.copy() for x in r.arrays(m)]
        _,n,met=cl.fit(aa,dict(server_round=1,expected_received_sha=r.c.digest(aa)))
        r.validate_stats(met,n);self.assertEqual(met['prototype_count'],4)
if __name__=='__main__':unittest.main(verbosity=2)
