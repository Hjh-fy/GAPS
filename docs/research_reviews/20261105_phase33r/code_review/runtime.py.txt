def encoder_sha(m):return c.digest([v.detach().cpu().numpy() for k,v in m.state_dict().items() if not k.startswith('classifier.')])
def strict_arrays(m,aa):
    state=list(m.state_dict().values())
    if len(aa)!=len(state):raise ValueError('state tensor count')
    for v,a in zip(state,aa):
        if tuple(v.shape)!=tuple(a.shape) or v.detach().cpu().numpy().dtype!=a.dtype or not np.isfinite(a).all():raise ValueError('state schema or finite mismatch')
def validate_state(m,state):
    reference=m.state_dict()
    if set(state)!=set(reference):raise ValueError('unknown model state keys')
    for k,v in reference.items():
        t=state[k]
        if not torch.is_tensor(t) or t.shape!=v.shape or t.dtype!=v.dtype or not torch.isfinite(t).all():raise ValueError('raw state schema/dtype/finite mismatch')
def validate_stats(metrics,n):
    if metrics.get('prototype_content_sha')!=__import__('hashlib').sha256(metrics.get('prototype_json','').encode()).hexdigest():raise ValueError('prototype content receipt mismatch')
    protos=json.loads(metrics.get('prototype_json','{}'));counts=json.loads(metrics.get('class_phase_counts_json','{}'));var=json.loads(metrics.get('prototype_var_json','{}'))
    keys={f'{g},0' for g in range(4)}
    if set(protos)!=keys or set(counts)!=keys or set(var)!=keys or any(type(v)!=int or v!=n//4 for v in counts.values()):raise ValueError('class stats identity/counts')
    for v in list(protos.values())+list(var.values())+[json.loads(metrics.get('global_feature_json','[]'))]:
        if np.asarray(v).shape!=(64,) or not np.isfinite(v).all():raise ValueError('stats feature/variance contract')
    if any((np.asarray(v)<0).any() for v in var.values()):raise ValueError('negative variance')
def load_checkpoint(m,path,lock,kind):
    c.checkpoint_manifest(path,lock,kind)
    ck=torch.load(path,map_location='cpu',weights_only=True)
    if set(ck)!={'model_state','kind','seed','source_identity_sha256'} or ck['kind']!=kind or ck['seed']!=42 or ck['source_identity_sha256']!=c.source_identity_sha():raise ValueError('checkpoint container')
    validate_state(m,ck['model_state']);m.load_state_dict(ck['model_state'],strict=True);strict_arrays(m,arrays(m))
    if kind=='sstar' and c.digest(arrays(m))!=lock['round_receipts'][-1]['returned']:raise ValueError('checkpoint endpoint mismatch')
    if kind=='random_source' and c.digest(arrays(m))!=lock['initial_sha256']:raise ValueError('random initial state mismatch')
    return m

class ReceiptMixin:
    def configure_fit(self,server_round,parameters,client_manager):
        rnd,params,manager=server_round,parameters,client_manager
        self.broadcast=c.digest(parameters_to_ndarrays(params))
        if rnd!=len(self.receipts)+1 or rnd not in range(1,26) or self.broadcast!=(self.receipts[-1]['returned'] if self.receipts else self.initial_state_sha):raise ValueError('server distribution chain mismatch')
        fits=super().configure_fit(rnd,params,manager)
        for proxy,ins in fits:ins.config.update(server_round=rnd,expected_received_sha=self.broadcast)
        self.protobuf_down=sum(serde.fit_ins_to_proto(ins).ByteSize() for _,ins in fits)
        return fits
    def aggregate_fit(self,server_round,results,failures):
        rnd=server_round
        if failures:raise ValueError('client failure;no partial aggregation')
        want=[1,2,5] if self.s['group']=='B2' else [1,2]
        ordered=sorted(results,key=lambda r:int(r[1].metrics.get('client_id',-1)))
        if [int(r.metrics.get('client_id',-1)) for _,r in ordered]!=want:raise ValueError('unexpected participants')
        samples=[]
        for _,r in ordered:
            cid=int(r.metrics['client_id']);aa=parameters_to_ndarrays(r.parameters);strict_arrays(self.schema_model,aa)
            if r.num_examples!={1:840,2:1680,5:32}[cid] or r.metrics.get('received_sha')!=self.broadcast or r.metrics.get('uploaded_sha')!=c.digest(aa):raise ValueError('upload identity/version/weight')
            if self.s['stats_upload']:
                validate_stats(r.metrics,r.num_examples)
                task.set_parameters(self.schema_model,aa,self.parameter_keys)
                if r.metrics.get('prototype_encoder_sha')!=encoder_sha(self.schema_model) or r.metrics.get('uploaded_encoder_sha')!=encoder_sha(self.schema_model):raise ValueError('stale prototype encoder')
            samples.append(r.num_examples)
        params,metrics=super().aggregate_fit(rnd,ordered,failures);returned=c.digest(parameters_to_ndarrays(params))
        self.receipts.append(dict(round=rnd,received=self.broadcast,returned=returned,clients=want,samples=samples,uploads=[r.metrics['uploaded_sha'] for _,r in ordered],prototype_encoder=[r.metrics.get('prototype_encoder_sha') for _,r in ordered],state_payload_up=sum(sum(a.nbytes for a in parameters_to_ndarrays(r.parameters)) for _,r in ordered),state_payload_down=len(want)*sum(a.nbytes for a in parameters_to_ndarrays(params)),protobuf_fit_up=sum(serde.fit_res_to_proto(r).ByteSize() for _,r in ordered),protobuf_fit_down=self.protobuf_down))
        self.receipts[-1].update(checkpoint_sha256=c.sha(self.output_dir/f'server_round_{rnd:03}.pth'),entry_sha256=c.sha(Path(__file__)),contract_sha256=c.sha(c.HERE/'contract.py'))
        c.write(self.output_dir/f'phase32_receipt_{rnd:03}.json',self.receipts[-1])
        return params,metrics

class PhaseGaps(ReceiptMixin,GapsStrategy):

    def _init_domain_adapt_loaders(self):

        self._calib_loader=c.target_loader(self.target_ds,self.current_round,self.target_audit)
    def _run_domain_adapt(self,rnd,state,aa,results,weights):
        self.current_round=rnd;start=len(self.target_audit);ss=len(self.source_audit)
        # Native reset_round_state refreshes model/optimizer. Refreshing loaders
        # each round binds the registered target block without changing the loop.
        self._init_domain_adapt_loaders()
        if self._da_trainer is not None:self._da_trainer.val_loader=self._val_loader;self._da_trainer.calib_loader=self._calib_loader
        result=super()._run_domain_adapt(rnd,state,aa,results,weights)
        used=self.target_audit[start:]

        return result
