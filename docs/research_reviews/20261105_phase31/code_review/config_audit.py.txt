"""Parse real archived CLIs without invoking main, model construction or training."""
from common import *

def plain(x):
    if isinstance(x,(str,int,float,bool,type(None))):return x
    if isinstance(x,(list,tuple)):return [plain(v) for v in x]
    if isinstance(x,dict):return {str(k):plain(v) for k,v in x.items()}
    return str(x)

def main():
    torch,cfg,ds,ns=runtime();out=HERE/'binding';out.mkdir(exist_ok=False)
    sp,se=parser_for('server',ns);cp,ce=parser_for('client',ns)
    (out/'SERVER_CLI_HELP.txt').write_text(sp.format_help(),encoding='utf-8')
    (out/'CLIENT_CLI_HELP.txt').write_text(cp.format_help(),encoding='utf-8')
    profiles=[]
    for profile in ns['CLASSIFICATION_PROFILE_FLAGS']:
        conf=ns['make_config'](profile=profile,batch_size=16,local_epochs=1,seed=42,num_classes=4,input_dim=8,num_clients=6,num_phases=3)
        profiles.append(dict(profile=profile,**{k:getattr(conf,k) for k in ['UPLOAD_PROTO_STATS','USE_ALIGN','USE_REPLAY_DISTILL','USE_PROTO_DECOUPLING','USE_REG_LOSS']}))
    js(out/'PROFILE_TRUTH_TABLE.json',profiles)
    structure=['--profile','ce_only','--seed','42','--num-classes','4','--input-dim','8','--num-clients','6','--num-phases','3']
    zero=['coral','global-mmd','class-mmd','proto-anchor','adv','proto','consistency','residual','proto-mmd','stage-mmd','align-reg-legacy']
    binds=[]
    for b in range(6):
        label=f'P3-PILOT-C5-A4-B{b}-S42'
        serverargv=['--strategy','gaps' if b>=3 else 'fedavg','--rounds','25','--min-clients','3' if b==2 else '2','--output-dir',f'UNRUN/{label}','--run-name',label]+structure

        for k in zero:serverargv+=['--da-lambda-'+k,'0']
        if b>=3:serverargv+=['--server-val-data',str(HERE/'data/client_1')+','+str(HERE/'data/client_2'),'--server-calib-data',str(HERE/'data/client_5')]
        args=sp.parse_args(serverargv);preset=se['apply_da_preset'](args,set(vars(args)))
        # The exact make_config arguments in server_app.main include batch_size=32.
        conf=ns['make_config'](seed=args.seed,num_classes=args.num_classes,input_dim=args.input_dim,num_clients=args.num_clients,num_phases=args.num_phases,batch_size=32,profile=args.profile)
        clients=[]
        for cid in ([] if b==1 else [1,2]+([5] if b==2 else [])):
            argv=['--client-id',str(cid),'--data-root',str(HERE/'data'/f'client_{cid}'),'--local-epochs','50' if cid==5 else '1','--batch-size','16','--eval-split','calibration']+structure

            cc=ns['make_config'](seed=ca.seed,num_classes=ca.num_classes,input_dim=ca.input_dim,num_clients=ca.num_clients,num_phases=ca.num_phases,batch_size=ca.batch_size,local_epochs=ca.local_epochs,profile=ca.profile,device=ca.device)
            clients.append(dict(client_id=cid,argv=argv,effective_args=vars(ca),effective_config=plain(vars(cc))))
        blockers=['no_proven_legal_Sstar','generic_server_has_no_initial_checkpoint_CLI','SEQ_LEN_100_vs_export_50']
        if b==1:blockers+=['no_generic_target_only_execution_entry;strict_target_entry_rejects_new_spec']
        if b>=3:blockers+=['server_batch32_causes_80000_target_presentations','pure_CE_profile_disables_stats_upload','server_source_pool_capped_500_not_full_2520']

    b3,b4,b5=[r['parsed_server_reference_even_if_not_used'] for r in binds[3:]]

    evidence=[]

    for rel in ['config.py','gaps_flower/server_app.py','gaps_flower/client_app.py','gaps_flower/task.py','gaps_flower/strategy.py','outputs/FedOnGas_严格少标定与系统成本_20261020/code_snapshot/gaps_flower/fedongas_strict_entry.py','outputs/FedOnGas_最终机制消融_C5_5pct_20261025/code/frozen_v2/matched_entry.py']:
        for ln,line in enumerate(textof(rel).splitlines(),1):
            if any(k in line for k in keywords):evidence.append(dict(path=rel,line=ln,code=line.strip()))
