class Capture:
    def __init__(self,m,torch):self.m=m;self.torch=torch;self.data={};self.order=[];self.handles=[]
    def save(self,name,value):
        if self.torch.is_tensor(value):
            a=value.detach().cpu().numpy().copy();assert a.ndim>=2 and a.dtype==np.float32 and np.isfinite(a).all()
            if name not in self.data:self.data[name]=[];self.order.append(name)
            self.data[name].append(a)
        elif isinstance(value,(tuple,list)):
            for i,v in enumerate(value):self.save(name+':'+str(i),v)
    def __enter__(self):
        self.handles.append(self.m.register_forward_pre_hook(lambda m,a:self.save('MODEL_INPUT',a[0])))
        for name,mod in self.m.named_modules():
            self.handles.append(mod.register_forward_hook(lambda m,a,o,n=name:self.save(n or 'MODEL_OUTPUT',o)))
        return self
    def __exit__(self,*args):
        for h in self.handles:h.remove()
    def arrays(self):return {k:np.concatenate(self.data[k]) for k in self.order}
