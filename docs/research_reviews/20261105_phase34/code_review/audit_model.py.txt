access=[]
allowed={str((c.DATA_ROOT/'source'/f'client_{cid}'/f'train_{n}.npy').resolve()).lower() for cid in [1,2] for n in ['features','classification_labels','phase_labels']}
def guard(event,args):
    if event!='open' or not isinstance(args[0],(str,bytes)):return
    p=Path(os.fsdecode(args[0]));s=str(p.resolve()).lower()
    if p.suffix.lower() in ['.npy','.npz']:
        access.append(dict(path=str(p),allowed=s in allowed))
        if s not in allowed:raise PermissionError('restore audit rejected non-source numeric input')
sys.addaudithook(guard)

    identities=c.prior.rows(c.CANONICAL_CONTRACTS/'SOURCE_WINDOW_IDENTITIES.csv');sourceids={cid:[x for x in identities if int(x['client_id'])==cid] for cid in [1,2]}
    rounds=[];files=[];previous=reg['initial_state_sha256'];max_aggregate_error=0.;all_steps=all_rows=0

            files.append(dict(role=f'client_{cid}_trained_snapshot',round=rnd,path=snap.relative_to(ROOT).as_posix(),sha256=c.sha(snap),state_sha256=native['uploaded'],bytes=snap.stat().st_size))
        # Independent high-precision ordinary weighted mean reference. Tolerance
