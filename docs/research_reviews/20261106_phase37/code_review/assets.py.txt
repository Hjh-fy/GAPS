ACCESS=[];HASH_ONLY=[False];GUARD_INSTALLED=False

def install_guard(lock):
    global GUARD_INSTALLED
    if GUARD_INSTALLED:return
    raw={str(Path(p).resolve()).lower() for p in lock['raw_Calibration_allowlist']}
    arrays={str(Path(p).resolve()).lower() for p in [lock['calibration_pack'],lock['classification_calibration_pack']]}
    def guard(event,args):
        if event!='open' or not isinstance(args[0],(str,bytes,os.PathLike)):return
        p=Path(os.fsdecode(args[0]));s=str(p.resolve()).lower();normalized=s.replace('\\','/')
        if '/dataset/data1/' in normalized or p.suffix.lower() in ['.npy','.npz']:
            allowed=s in raw or s in arrays or HASH_ONLY[0]
            ACCESS.append(dict(path=str(p),allowed=allowed,purpose='SHA_ONLY' if HASH_ONLY[0] else 'CALIBRATION_NUMERIC'))
            if not allowed:raise PermissionError('P37 blocks all non-Calibration numeric input BEFORE OPEN: '+str(p))
        if '/fedongas_phase35_20261105/evaluation/' in normalized and not HASH_ONLY[0]:raise PermissionError('P37 excludes Phase35 Test output consumption')
    sys.addaudithook(guard);GUARD_INSTALLED=True

def hash_only(p):
    HASH_ONLY[0]=True
    try:return sha(p)
    finally:HASH_ONLY[0]=False

class ClassifierAdapter:
    def __init__(self,group,model,r,version,state_version):self.group=group;self.model=model;self.r=r;self.version=version;self.state_version=state_version
    def classify(self,window):
        with self.r.torch.inference_mode():logits=self.model(self.r.torch.from_numpy(window[None,:]))[0][0].cpu().numpy()
        return dict(logits=logits,predicted_class=int(np.argmax(logits)),version=self.version)

class ExtractorAdapter:
    def __init__(self,v,lock):self.v=v;self.names=tuple(lock['feature_contract']['names']);self.version=lock['feature_contract_sha256']
    def extract(self,window):
        # EXACT archived phase1 formula. Metadata and physical gas are absent.
        vector=self.v.features.extract_canonical_features(window,phase=-1,metadata={},client='unknown',split='strict_phase1',sample_index=0).sensor83[self.v.idx]
        return dict(values=vector,names=self.names,version=self.version)

class SVRAdapter:
    def __init__(self,h,v,version):self.h=h;self.v=v;self.version=version;self.gas_id=h['gas'];self.support=(float(h['clip_min']),float(h['clip_max']))
    def predict_raw(self,x):return self.v.svr.predict_head(self.h['numerical'],x,clip=False)
    def scaler_sha(self):return hashlib.sha256(np.ascontiguousarray(self.h['numerical']['mean']).tobytes()+np.ascontiguousarray(self.h['numerical']['scale']).tobytes()).hexdigest()
